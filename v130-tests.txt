/* בדיקה לגרסה 130 — 19.10: קישור מהתזכורת (?lookupPhone=…) — "מלא קפיצות עד
 * שרואים את התור". סופרים קפיצות: גלילות (scrollY משתנה) ותזוזות תצוגה (layout
 * shift, שהדפדפן רושם מרגע פתיחת הדף) — מהפתיחה ועד שהתור על המסך. */
window.V30 = (function(){
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const result = (name, pass, failMsg, detail) => ({ name: name, pass: !!pass, msg: pass ? 'עבר' : failMsg, detail: detail });
  return {
    /* הכנה: "ביקור קודם" עם לוגו שמתעכב (כתובת שלא עונה) */
    prepareSlowLogo(){
      localStorage.setItem('hbk.theme.v1', JSON.stringify({ vars: { '--th-primary': '#f2a396' }, classes: 'th-on th-arc', name: 'סלון בדיקה', hasLogo: true,
        logo: 'http://10.255.255.1/slow-logo.png', logoFill: true, logoZoom: 1 }));
      return 'ok';
    },
    /* השם לא מופיע לפני הלוגו */
    async nameWaitsForLogo(){
      await sleep(Math.max(0, 1200 - performance.now()));
      const name = document.getElementById('heroBizName');
      const med = document.getElementById('heroMedallion');
      const nameVisible = !!(name && name.classList.contains('biz-name-visible') && getComputedStyle(name).visibility !== 'hidden');
      const logoShown = !!(med && med.classList.contains('has-logo'));
      return result('השם לא מופיע לפני הלוגו (מגיעים יחד)', !(nameVisible && !logoShown),
        'שם הסלון מופיע לפני הלוגו', { atMs: Math.round(performance.now()), nameVisible, logoShown, medCls: med && med.className });
    },
    async linkWithoutJumps(){
      const shifts = [];
      try { new PerformanceObserver(l => l.getEntries().forEach(e => { if (!e.hadRecentInput) shifts.push(Math.round(e.value * 1000) / 1000); })).observe({ type: 'layout-shift', buffered: true }); } catch (e) {}
      const ys = [window.scrollY];
      let ready = false;
      for (let i = 0; i < 150 && !ready; i++){
        if (window.scrollY !== ys[ys.length - 1]) ys.push(window.scrollY);
        const m = document.getElementById('myAppointment');
        ready = !!(m && /נמצא תור/.test(m.textContent || '') && !document.documentElement.classList.contains('hk-deeplink-wait'));
        await sleep(50);
      }
      await sleep(400);
      if (window.scrollY !== ys[ys.length - 1]) ys.push(window.scrollY);
      const m = document.getElementById('myAppointment');
      const top = m ? Math.round(m.getBoundingClientRect().top) : null;
      const bigShifts = shifts.filter(v => v >= 0.05).length;
      return result('קישור לתורים: התור בראש המסך בלי קפיצות (גלילות ותזוזות גדולות)', ready && top !== null && top >= -5 && top < window.innerHeight * 0.5 && ys.length <= 2 && bigShifts === 0,
        'עד שהתור מופיע הדף קופץ כמה פעמים', { ready, top, scrollPositions: ys, layoutShifts: shifts, bigShifts });
    },
  };
})();
